type CycleBoxObj
    int startBar
    int endBar
    int tier
    box b = na

method deleteCycle(CycleBoxObj this) =>
    if not na(this.b)
        box.delete(this.b)
    this.b := na
    int(na)

type HMCWindowBox
    int startBar
    int endBar
    float hi
    float lo
    string title
    color col
    box b = na
    label lbl = na

method deleteHMC(HMCWindowBox this) =>
    if not na(this.b)
        box.delete(this.b)
    if not na(this.lbl)
        label.delete(this.lbl)
    this.b   := na
    this.lbl := na
    int(na)

method updateHMC(HMCWindowBox this, string sz) =>
    if na(this.b)
        this.b   := box.new(this.startBar, this.hi, this.endBar, this.lo, xloc=xloc.bar_index, bgcolor=this.col, border_color=color.new(this.col, 40), border_width=1)
        this.lbl := label.new(this.startBar, this.hi, this.title, xloc=xloc.bar_index, style=label.style_none, textcolor=#2a2e39, size=sz)
    if not na(this.b)
        box.set_right(this.b, this.endBar)
        box.set_top(this.b, this.hi)
        box.set_bottom(this.b, this.lo)
        label.set_xy(this.lbl, int(math.avg(this.startBar, this.endBar)), this.hi)
        label.set_size(this.lbl, sz)
    int(na)

f_findMSNRConfluence(float zTop, float zBot) =>
    string conf = ""
    if showMSNRLines and showQML and qmStage >= 3 and not na(qmLsP)
        if qmLsP <= zTop and qmLsP >= zBot
            conf := (qmDir == -1 ? "QMLSELL " : "QMLBUY ") + f_prettyTF(tf_Chart)
    if conf == ""
        for s in msnrSources
            if s.levelPrice <= zTop and s.levelPrice >= zBot
                conf := s.tfTag + " " + s.kind
                break
    if conf == ""
        for o in msnrOCLs
            if o.levelPrice <= zTop and o.levelPrice >= zBot
                conf := o.tfTag + " " + o.kind
                break
    conf

f_symPrefix(string tierName, bool isSellSide) =>
    basePfx = str.startswith(zoneNamingStyle, "FW") ? ("FW " + tierName) : ("GAMBLER " + tierName)
    sideTag = isSellSide ? " (SELL)" : " (BUY)"
    basePfx + sideTag

f_createSymZone(int sBar, float mult1, float mult2, float rHi, float rSpan, bool isSellZone, string tierTitle, color zCol, bool hasMidLine) =>
    p1     = rHi - (mult1 * rSpan)
    p2     = rHi - (mult2 * rSpan)
    zTop   = math.max(p1, p2)
    zBot   = math.min(p1, p2)
    zMid   = (zTop + zBot) * 0.5
    invP   = isSellZone ? zTop : zBot
    pxInfo = showSymPrices ? (" [" + str.tostring(zBot, format.mintick) + " - " + str.tostring(zTop, format.mintick) + "]  ") : "  "
    lbl    = f_symPrefix(tierTitle, isSellZone) + pxInfo
    brdCol = isSellZone ? color.new(#e53935, 45) : color.new(#1e88e5, 45)
    bx     = box.new(sBar, zTop, bar_index + 4, zBot, xloc=xloc.bar_index, bgcolor=zCol, border_color=brdCol, border_width=1, text=lbl, text_color=#1e222d, text_size=txtSize, text_halign=text.align_right, text_valign=text.align_center)
    line mLn = hasMidLine ? line.new(sBar, zMid, bar_index + 4, zMid, xloc=xloc.bar_index, color=brdCol, style=line.style_dotted, width=1) : na
    array.push(gamblerZones, GamblerZone.new(sBar, zTop, zBot, zMid, invP, isSellZone, true, lbl, zCol, bx, mLn))
    int(na)
